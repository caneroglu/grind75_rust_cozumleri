impl Solution {

    
    pub fn two_sum(nums: Vec<i32>, target: i32) -> Vec<i32> {

        use std::collections::HashMap;

        let mut sol: HashMap<i32, i32> = HashMap::new();
        for (indx,&val) in nums.iter().enumerate() {
            let diff = target - val;
            match sol.get(&diff){
                Some(second_indx) => return vec![*second_indx, indx as i32],
                None => sol.insert(val, indx as i32),
            };
        }

     vec![0,0]


    }
}

impl Solution {

    
    pub fn two_sum(nums: Vec<i32>, target: i32) -> Vec<i32> {

        use std::collections::HashMap;
        let mut sol: HashMap<i32, usize> = HashMap::new();

        let mut ans: Vec<i32> = Vec::new();
        for (indx,&val) in nums.iter().enumerate() {
            
            let diff = target - val;

            if sol.contains_key(&diff) {

                ans.push(*(sol.get(&diff).unwrap()) as i32);
                ans.push(indx as i32);

               break;
            
            };
            sol.insert(val, indx);
        }

        ans


    }
}
